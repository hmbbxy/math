## 双重退火
```python
from scipy.optimize import dual_annealing
import numpy as np
import matplotlib.pyplot as plt

# 用来保存每一步迭代的目标函数值，画图用
history = []

# 目标函数 + 多个约束（惩罚函数法）
def objective_penalty(vars):
    x, y = vars
    # 原始目标函数：最小化 f = x² + y² +5*sin(x)
    f = x**2 + y**2 + 5 * np.sin(x)

    # ========== 多个约束条件 ==========
    # 约束1：x + y >= 2
    cons1 = max(0, 2 - (x + y))
    # 约束2：x >= 0
    cons2 = max(0, 0 - x)
    # 约束3：y <= 8
    cons3 = max(0, y - 8)

    # 惩罚系数，约束不满足就加很大的惩罚值
    penalty_coeff = 1000
    total_penalty = penalty_coeff * (cons1 + cons2 + cons3)
    total_cost = f + total_penalty

    # 记录每一步的函数值，用于绘图
    history.append(total_cost)
    return total_cost

# 变量取值范围
bounds = [(-10, 10), (-10, 10)]

# 双重退火求解
result = dual_annealing(objective_penalty, bounds=bounds)

# 输出结果
print("===== 优化结果 =====")
print("最优变量 x,y：", result.x)
print("目标函数最小值：", result.fun)
print("迭代次数：", result.nit)
print("收敛成功？", result.success)
print("信息：", result.message)

# ========== 绘制收敛曲线 ==========
plt.rcParams['font.sans-serif'] = ['SimHei'] # 显示中文
plt.rcParams['axes.unicode_minus'] = False
plt.figure(figsize=(8,4))
plt.plot(history, c='#2E86AB')
plt.xlabel("迭代次数")
plt.ylabel("目标函数值（含惩罚）")
plt.title("双重退火算法收敛曲线")
plt.grid(alpha=0.3)
plt.tight_layout()
plt.show()
```

+ 逐段关键说明
 
1. history列表：每次计算目标函数时，把当前值存入，用来画出收敛图，论文图可以直接放进附录/结果图。
2. 惩罚项原理
 max(0, 约束违反量) ：只有不满足约束的时候才会产生惩罚；满足约束时惩罚=0，不影响目标函数。
 penalty_coeff ：惩罚系数，越大约束越严格，一般取100~10000，根据你的模型调。
 
注意：惩罚系数不能无限大，太大会导致数值难收敛。
 
3. dual_annealing特点
 
- 不需要给初始猜测点，自动在bounds区间全局搜索
- 适合多峰非线性优化，不容易卡在局部最优
- 原生不支持直接写约束，惩罚函数是数模最常用方案

## 双重退火 + 批量灵敏度分析完整代码
 
逻辑：修改约束条件的阈值，多次运行双重退火，保存每组参数对应的最优解与目标值，输出结果表，可直接用于数模论文灵敏度分析
 
 ```python
  
from scipy.optimize import dual_annealing
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# 全局列表保存所有批次结果
result_list = []

# 定义函数：接收约束阈值，执行一次优化
def run_optimization(threshold):
    history = []
    def objective_penalty(vars):
        x, y = vars
        f = x**2 + y**2 + 5 * np.sin(x)
        # 约束1：x + y >= threshold（这个是我们要做灵敏度分析的参数）
        cons1 = max(0, threshold - (x + y))
        # 约束2：x >= 0
        cons2 = max(0, 0 - x)
        # 约束3：y <= 8
        cons3 = max(0, y - 8)
        penalty_coeff = 1000
        total_penalty = penalty_coeff * (cons1 + cons2 + cons3)
        total_cost = f + total_penalty
        history.append(total_cost)
        return total_cost

    bounds = [(-10, 10), (-10, 10)]
    res = dual_annealing(objective_penalty, bounds=bounds)
    return res

# ========== 灵敏度参数设置：遍历不同阈值 ==========
# 测试 x+y >= threshold，阈值从1.5逐步增加到3.0
threshold_list = np.linspace(1.5, 3.0, 8) ## 生成8个数

for t in threshold_list:
    opt_res = run_optimization(t)
    x_opt, y_opt = opt_res.x
    result_list.append({
        "约束阈值t": t,
        "最优x": x_opt,
        "最优y": y_opt,
        "目标函数最小值": opt_res.fun,
        "收敛标记": opt_res.success
    })

# 转为DataFrame，输出表格
df = pd.DataFrame(result_list)
print("========灵敏度分析结果表========")
print(df.round(4))

# ========== 绘图：约束阈值变化，目标最优值变化 ==========
plt.rcParams['font.sans-serif'] = ['SimHei']
plt.rcParams['axes.unicode_minus'] = False
plt.figure(figsize=(8,4))
plt.plot(df["约束阈值t"], df["目标函数最小值"], marker='o', color='#c82423')
plt.xlabel("约束阈值 t (x+y ≥ t)")
plt.ylabel("最优目标函数值")
plt.title("灵敏度分析：约束阈值对最优目标值的影响")
plt.grid(alpha=0.3)
plt.tight_layout()
plt.show()
```
